const canvas = document.getElementById('matrix');

function resize() {
  canvas.width = window.innerWidth;

let columns = Math.floor(canvas.width / fontSize);
let drops = new Array(columns).fill(1);

function drawMatrix() {
  ctx.fillStyle = "rgba(0,0,0,0.08)";
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = "#00ff41";
  ctx.font = fontSize + "px monospace";

    const text = chars[Math.floor(Math.random() * chars.length)];
    ctx.fillText(text, i * fontSize, drops[i] * fontSize);
    if (drops[i] * fontSize > canvas.height && Math.random() > 0.975) {
      drops[i] = 0;
    }
    drops[i]++;
  }
}

// ---- Terminal sequence ----
const term = document.getElementById('terminal');

function randIP() {
  return `${rnd(1,255)}.${rnd(0,255)}.${rnd(0,255)}.${rnd(1,255)}`;
}
function randHex(len) {
  let s = '';
  const chars = '0123456789ABCDEF';
  for (let i=0;i<len;i++) s += chars[Math.floor(Math.random()*chars.length)];
  return s;
}
function randMac() {
  let parts = [];
  for (let i=0;i<6;i++) parts.push(randHex(2));
  return parts.join(':');
}
function rnd(min,max) { return Math.floor(Math.random()*(max-min+1))+min; }
function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

async function typeLine(html, opts = {}) {
  const speed = opts.speed || 8;
  const line = document.createElement('div');
  term.appendChild(line);
  const plain = html.replace(/<[^>]*>/g, '');
  let shown = '';
  for (let i = 0; i < plain.length; i++) {
    shown += plain[i];
    line.textContent = shown;
    term.scrollTop = term.scrollHeight;
    if (!opts.instant) await sleep(speed + Math.random()*10);
  }
  if (opts.cls) line.className = opts.cls;
  await sleep(opts.pause || 60);
  return line;
}

function printLine(html, cls) {
  const line = document.createElement('div');
  line.innerHTML = html;
  if (cls) line.className = cls;
  term.appendChild(line);
  term.scrollTop = term.scrollHeight;
}

async function progressBar(label, ms, opts = {}) {
  const line = document.createElement('div');
  term.appendChild(line);
  const steps = 24;
  const stepTime = ms / steps;
  for (let i = 0; i <= steps; i++) {
    const filled = '█'.repeat(i);
    const empty = '░'.repeat(steps - i);
    const pct = Math.floor((i/steps)*100);
    line.innerHTML = `${label} <span class="bar-track">[${filled}${empty}]</span> ${pct}%`;
    term.scrollTop = term.scrollHeight;
    await sleep(stepTime);
  }
  await sleep(opts.pause || 150);
}

async function fakeFileScan(count, dir) {
  const exts = ['.dat','.cfg','.db','.log','.bak','.key','.sys','.tmp'];
  for (let i=0;i<count;i++) {
    const name = randHex(rnd(4,8)).toLowerCase();
    const ext = exts[rnd(0,exts.length-1)];
    const size = rnd(1,900);
    printLine(`<span class="dim">${dir}/${name}${ext}</span> <span class="dim">... ${size}KB</span>`);
    term.scrollTop = term.scrollHeight;
    await sleep(rnd(15,45));
  }
}
